# 问题完整分析 + 修复方案
## 一、报错根源：`Response remained truncated after 3 continuation attempts`
两个核心诱因叠加：
1. **代码diff日志重复刷屏**
   旧代码里连续十几行重复 `self.log_message(f"✓ custom→function 转换 #{Handler.count}")`，日志输出疯狂膨胀，返回报文超长截断；
2. **http.client 转发逻辑未完整补全**
   切换 `http.client.HTTPSConnection` 替代 urllib 后，diff 省略了后半段收发响应、异常捕获、回传给客户端的完整代码，代理逻辑残缺，请求转发中途断流，连续3次续读都拿不到完整响应直接抛截断错误；
3. 附加隐患：
   - 重复日志撑满输出缓冲区；
   - 缺少 `conn.close()` 连接回收，长连接堆积导致端口15730数据流阻塞；
   - 异常分支没有统一返回500错误给前端，半响应直接断开。

## 二、完整修复版 codex_tool_proxy.py（v4 无冗余日志 + 完整HTTPS转发）
```python
"""
Codex MCP工具类型转换代理 v4
解决 MiMo API 不支持 Responses API 中 custom 工具类型的问题
用 http.client 替代 urllib，精确控制请求发送
"""
import json
import sys
import re
import os
import argparse
import ssl
from http.server import HTTPServer, BaseHTTPRequestHandler
from http.client import HTTPSConnection
from urllib.parse import urlparse

PORT = 15730
UPSTREAM = "https://token-plan-cn.xiaomimimo.com"
# 从环境变量读取API Key，无则运行时报错
MIMO_API_KEY = os.getenv("MIMO_API_KEY", "")

def convert_tools(tools_list):
    """将type:custom工具转为type:function标准格式"""
    new_tools = []
    changed = False
    for tool in tools_list:
        if tool.get("type") == "custom":
            changed = True
            func_def = tool.get("custom", {})
            new_tool = {
                "type": "function",
                "function": func_def
            }
            new_tools.append(new_tool)
        else:
            new_tools.append(tool)
    return new_tools, changed

class ProxyHandler(BaseHTTPRequestHandler):
    count = 0
    api_key = MIMO_API_KEY

    def _proxy(self, method):
        length = int(self.headers.get("Content-Length", 0))
        body = self.rfile.read(length) if length else b""

        # 尝试转换工具类型
        modified = False
        try:
            data = json.loads(body.decode("utf-8"))
            if "tools" in data and isinstance(data["tools"], list):
                new_tools, changed = convert_tools(data["tools"])
                if changed:
                    data["tools"] = new_tools
                    body = json.dumps(data, ensure_ascii=False).encode("utf-8")
                    modified = True
                    ProxyHandler.count += 1
                    self.log_message(f"✓ custom→function 转换 #{ProxyHandler.count}")
        except Exception as e:
            self.log_message(f"工具转换跳过，解析异常: {str(e)}")

        # 解析上游地址
        parsed = urlparse(UPSTREAM)
        host = parsed.hostname
        port = parsed.port or (443 if parsed.scheme == "https" else 80)
        path = self.path

        self.log_message(f"→ {method} {path} → {host}:{port}")

        # 构建请求头
        req_headers = {
            "Content-Type": "application/json; charset=utf-8",
            "Content-Length": str(len(body)),
            "Authorization": f"Bearer {self.api_key}",
            "Host": host,
        }
        # 保留原始Accept等头，过滤冲突头
        for k, v in self.headers.items():
            kl = k.lower()
            if kl not in ("host", "content-length", "content-type", "authorization", "connection"):
                req_headers[k] = v

        conn = None
        try:
            # 创建HTTPS连接
            ctx = ssl.create_default_context()
            conn = HTTPSConnection(host, port=port, context=ctx, timeout=120)
            conn.request(method, path, body=body, headers=req_headers)
            resp = conn.getresponse()

            # 回写状态码+响应头给客户端
            self.send_response(resp.status)
            for hk, hv in resp.getheaders():
                lhk = hk.lower()
                if lhk not in ("transfer-encoding", "content-length"):
                    self.send_header(hk, hv)
            self.end_headers()

            # 流式回传响应体，避免大报文截断
            chunk_size = 4096
            while chunk := resp.read(chunk_size):
                self.wfile.write(chunk)
            return

        except Exception as err:
            err_msg = f"上游API转发失败: {str(err)}"
            self.log_message(f"× {err_msg}")
            err_body = json.dumps({"error": err_msg}).encode("utf-8")
            self.send_response(500)
            self.send_header("Content-Type", "application/json")
            self.send_header("Content-Length", str(len(err_body)))
            self.end_headers()
            self.wfile.write(err_body)
        finally:
            if conn:
                conn.close()

    def do_POST(self):
        self._proxy("POST")

    def do_GET(self):
        self._proxy("GET")

    def log_message(self, format, *args):
        # 统一日志输出，屏蔽原生冗余访问日志
        sys.stderr.write(f"[PROXY] {format % args}\n")

def main():
    parser = argparse.ArgumentParser(description="Codex MCP Tool Proxy v4")
    parser.add_argument("--port", type=int, default=PORT, help="监听端口")
    parser.add_argument("--api-key", type=str, help="MiMo API Key，优先环境变量MIMO_API_KEY")
    args = parser.parse_args()

    global MIMO_API_KEY
    if args.api_key:
        MIMO_API_KEY = args.api_key
    if not MIMO_API_KEY:
        print("错误：必须设置MIMO_API_KEY环境变量或传入--api-key参数")
        sys.exit(1)

    server_addr = ("127.0.0.1", args.port)
    httpd = HTTPServer(server_addr, ProxyHandler)
    print(f"Codex Tool Proxy v4 启动成功，监听 http://127.0.0.1:{args.port}")
    print(f"上游目标: {UPSTREAM}")
    try:
        httpd.serve_forever()
    except KeyboardInterrupt:
        print("\n代理服务正常退出")
        httpd.shutdown()

if __name__ == "__main__":
    main()
```

## 三、关键修复点说明
1. **删除重复刷屏日志**
   移除十几行重复 `self.log_message`，只保留1行转换日志，彻底解决日志超长截断；
2. **补全完整 http.client 收发逻辑**
   - 建立连接、发送请求、读取响应、流式回传给本地客户端；
   - `finally` 强制关闭HTTPS连接，杜绝端口长连接泄漏；
3. **大报文流式传输**
   采用4096字节分片读取上游响应，一次性读取超大JSON会造成缓冲区溢出截断；
4. **完善异常捕获**
   转发失败统一返回500 JSON错误，不会出现半断流无响应；
5. **命令行参数兼容**
   支持 `--api-key` 临时传密钥，兼容环境变量部署；
6. 过滤冲突响应头
   自动过滤 `transfer-encoding` 避免本地http.server解析冲突。
